cmake_minimum_required(VERSION 3.22)
project(moenotes-score-native VERSION 0.1.0 LANGUAGES C CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(MNS_BUILD_TESTS "Build synthetic tests" ON)
option(MNS_BUILD_CLI "Build JSON CLI" ON)
option(MNS_SANITIZE "Enable address and undefined behavior sanitizers" OFF)
option(MNS_BUILD_FUZZER "Build Clang libFuzzer input target" OFF)
set(MOENOTES_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(MOENOTES_BUILD_CLI OFF CACHE BOOL "" FORCE)
add_subdirectory(vendor/chart-parser EXCLUDE_FROM_ALL)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/vendor/boost/boost/multiprecision/cpp_int.hpp")
 set(MNS_BOOST_INCLUDE "${CMAKE_CURRENT_SOURCE_DIR}/vendor/boost")
else()
 find_path(MNS_BOOST_INCLUDE boost/multiprecision/cpp_int.hpp HINTS "${CMAKE_CURRENT_SOURCE_DIR}/../../tools/boost-1.83/usr/include" REQUIRED)
endif()
add_library(moenotes-score-native src/json.cpp src/prepared.cpp src/scoring.cpp src/power.cpp src/runtime.cpp src/runtime_machine.cpp src/analysis.cpp src/bounds.cpp src/api.cpp)
add_library(moenotes::score ALIAS moenotes-score-native)
target_include_directories(moenotes-score-native PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include> PRIVATE src vendor/chart-parser/third_party)
target_include_directories(moenotes-score-native SYSTEM PRIVATE ${MNS_BOOST_INCLUDE})
target_link_libraries(moenotes-score-native PRIVATE moenotes::chart_parser)
set_target_properties(moenotes-score-native PROPERTIES EXPORT_NAME score WINDOWS_EXPORT_ALL_SYMBOLS ON)
if(MSVC)
 target_compile_options(moenotes-score-native PRIVATE /W4 /fp:strict)
 target_compile_options(moenotes_chart_parser PRIVATE /fp:strict)
else()
 target_compile_options(moenotes-score-native PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off -fno-fast-math)
 if(MNS_SANITIZE)
  target_compile_options(moenotes-score-native PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(moenotes-score-native PUBLIC -fsanitize=address,undefined)
  target_compile_options(moenotes_chart_parser PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
 endif()
endif()
if(MNS_BUILD_CLI AND NOT EMSCRIPTEN)
 add_executable(moenotes-score cli/main.cpp)
 target_include_directories(moenotes-score PRIVATE src)
 target_link_libraries(moenotes-score PRIVATE moenotes-score-native)
endif()
if(EMSCRIPTEN)
 add_executable(moenotes-score-wasm web/entry.cpp)
 target_link_libraries(moenotes-score-wasm PRIVATE moenotes-score-native)
 set_target_properties(moenotes-score-wasm PROPERTIES OUTPUT_NAME moenotes-score SUFFIX .mjs)
 target_link_options(moenotes-score-wasm PRIVATE
  --no-entry -sMODULARIZE=1 -sEXPORT_ES6=1 -sENVIRONMENT=web,worker,node
  -sALLOW_MEMORY_GROWTH=1 -sMAXIMUM_MEMORY=1073741824 -sINITIAL_MEMORY=33554432
  -sDISABLE_EXCEPTION_CATCHING=0
  "-sEXPORTED_FUNCTIONS=['_malloc','_free','_mns_version','_mns_last_error','_mns_string_free','_mns_prepare','_mns_prepared_free','_mns_prepared_report','_mns_deck_resolve','_mns_deck_free','_mns_deck_report','_mns_analysis_create','_mns_analysis_free','_mns_analysis_advance','_mns_analysis_snapshot','_mns_analysis_cancel','_mns_analysis_resume']"
  "-sEXPORTED_RUNTIME_METHODS=['UTF8ToString','HEAPU8','HEAPU32']")
 target_compile_options(moenotes-score-native PRIVATE -fexceptions)
endif()
if(MNS_BUILD_TESTS AND NOT EMSCRIPTEN)
 enable_testing()
 foreach(name core power runtime runtime_machine analysis bounds api)
  if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/test_${name}.cpp")
   add_executable(test_${name} tests/test_${name}.cpp)
   target_include_directories(test_${name} PRIVATE src ${MNS_BOOST_INCLUDE})
   target_link_libraries(test_${name} PRIVATE moenotes-score-native)
   target_compile_definitions(test_${name} PRIVATE MNS_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}")
   if(MSVC)
    target_compile_options(test_${name} PRIVATE /UNDEBUG)
   else()
    target_compile_options(test_${name} PRIVATE -UNDEBUG -ffp-contract=off -fno-fast-math)
   endif()
   add_test(NAME ${name} COMMAND test_${name})
   set_tests_properties(${name} PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
  endif()
 endforeach()
 if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/tests/power_driver.cpp")
  add_executable(test_power_driver tests/power_driver.cpp)
  target_include_directories(test_power_driver PRIVATE src)
  target_link_libraries(test_power_driver PRIVATE moenotes-score-native)
 endif()
endif()
if(MNS_BUILD_FUZZER)
 if(NOT CMAKE_CXX_COMPILER_ID MATCHES "Clang")
  message(FATAL_ERROR "MNS_BUILD_FUZZER requires Clang")
 endif()
 target_compile_options(moenotes-score-native PRIVATE -fsanitize=fuzzer-no-link,address,undefined)
 target_compile_options(moenotes_chart_parser PRIVATE -fsanitize=fuzzer-no-link,address,undefined)
 add_executable(mns-fuzz-inputs tests/fuzz_inputs.cpp)
 target_link_libraries(mns-fuzz-inputs PRIVATE moenotes-score-native)
 target_compile_options(mns-fuzz-inputs PRIVATE -fsanitize=fuzzer,address,undefined)
 target_link_options(mns-fuzz-inputs PRIVATE -fsanitize=fuzzer,address,undefined)
endif()
install(TARGETS moenotes-score-native moenotes_chart_parser EXPORT moenotes-score-native-targets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES include/moenotes_score.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(FILES README.md VALIDATION.md THIRD_PARTY.md DESTINATION ${CMAKE_INSTALL_DOCDIR})
install(FILES vendor/boost/LICENSE_1_0.txt DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/moenotes-score-native/boost)
install(FILES vendor/chart-parser/LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/moenotes-score-native/chart-parser)
install(FILES vendor/chart-parser/third_party/LICENSE.yyjson DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/moenotes-score-native/yyjson)
install(FILES vendor/chart-parser/include/moenotes_chart_parser.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
set(MNS_CONFIG_DIR ${CMAKE_INSTALL_LIBDIR}/cmake/moenotes-score-native)
configure_package_config_file(cmake/moenotes-score-native-config.cmake.in ${CMAKE_CURRENT_BINARY_DIR}/moenotes-score-native-config.cmake INSTALL_DESTINATION ${MNS_CONFIG_DIR})
write_basic_package_version_file(${CMAKE_CURRENT_BINARY_DIR}/moenotes-score-native-config-version.cmake VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
install(EXPORT moenotes-score-native-targets NAMESPACE moenotes:: DESTINATION ${MNS_CONFIG_DIR})
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/moenotes-score-native-config.cmake ${CMAKE_CURRENT_BINARY_DIR}/moenotes-score-native-config-version.cmake DESTINATION ${MNS_CONFIG_DIR})
if(TARGET moenotes-score)
 install(TARGETS moenotes-score RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
endif()
